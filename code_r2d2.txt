// ===============================
// ПИНЫ И НАСТРОЙКИ РОБОТА
// ===============================
const int IN1 = 16;
const int IN2 = 17;
const int IN3 = 18;
const int IN4 = 19;

const int PIN_SERVO_HEAD  = 25;
const int PIN_SERVO_LEG_L = 26;
const int PIN_SERVO_LEG_R = 27;
const int PIN_SERVO_ARM   = 14;

const int PIN_LED_R = 32;
const int PIN_LED_G = 33;
const int PIN_LED_B = 23;
const int PIN_BUZZER = 4;

const int MPU_ADDR = 0x68;
const float IMPACT_THRESHOLD = 26000.0;

Servo servoHead;
Servo servoLegL;
Servo servoLegR;
Servo servoArm;

// Предварительное объявление функций
void moveRobot(int speedLeft, int speedRight);
void rotateHead(int angle);
void setBodyTilt(int angle);
void moveArm(int position);
void setLEDColor(int r, int g, int b);
void playSound(unsigned int frequency, unsigned long duration);
bool checkCollision();
void playEmotionShock();
bool waitAndCheck(unsigned long ms);
void danceRoutine();
void r2Chirp(int times);
void resetPosition();

// ===============================
// ОБРАБОТЧИКИ ВЕБ-СЕРВЕРА
// ===============================
void handleRoot() {
  server.send(200, "text/html", MAIN_PAGE);
}

void handleDanceOn() {
  isDancing = true;
  server.send(200, "text/plain", "OK");
}

void handleDanceOff() {
  isDancing = false;
  resetPosition(); // Немедленно останавливаем все механизмы
  server.send(200, "text/plain", "OK");
}

// ===============================
// SETUP
// ===============================
void setup() {
  Serial.begin(115200);

  // Настройка пинов
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  pinMode(PIN_LED_R, OUTPUT); pinMode(PIN_LED_G, OUTPUT); pinMode(PIN_LED_B, OUTPUT);
  pinMode(PIN_BUZZER, OUTPUT);

  // Сервоприводы
  servoHead.attach(PIN_SERVO_HEAD, 500, 2400);
  servoLegL.attach(PIN_SERVO_LEG_L, 500, 2400);
  servoLegR.attach(PIN_SERVO_LEG_R, 500, 2400);
  servoArm.attach(PIN_SERVO_ARM, 500, 2400);

  // I2C
  Wire.begin(21, 22);
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B); Wire.write(0x00);
  Wire.endTransmission();

  // Исходная позиция
  resetPosition();

  // Запуск Wi-Fi Точки Доступа
  Serial.println("\nНастройка Wi-Fi точки доступа...");
  WiFi.softAP(AP_SSID, AP_PASSWORD);
  IPAddress IP = WiFi.softAPIP();
  Serial.print("Точка доступа запущена! IP адрес: ");
  Serial.println(IP); // Обычно это 192.168.4.1

  // Маршруты сервера
  server.on("/", handleRoot);
  server.on("/dance/on", handleDanceOn);
  server.on("/dance/off", handleDanceOff);
  server.begin();
  
  delay(1000);
}

// ===============================
// LOOP
// ===============================
void loop() {
  server.handleClient(); // Постоянная обработка запросов от телефона

  if (isDancing) {
    danceRoutine(); 
  } else {
    // Даже если он стоит на месте, проверяем удары
    if (checkCollision()) {
      playEmotionShock();
    }
  }
}

// ===============================
// УМНАЯ ЗАДЕРЖКА (С ВЕБ-СЕРВЕРОМ И ДАТЧИКОМ)
// ===============================
bool waitAndCheck(unsigned long ms) {
  unsigned long start = millis();
  while (millis() - start < ms) {
    server.handleClient(); // Отвечаем телефону во время танца

    if (!isDancing) {
      return true; // Если нажали кнопку "СТОП" на телефоне — прерываем танец
    }
    
    if (checkCollision()) {
      playEmotionShock(); // Реакция на удар (грусть)
      return true;        // Прерываем текущий цикл танца, чтобы начать его заново
    }
    delay(10);
  }
  return false;
}

// ================= СЦЕНАРИЙ: БОЛЬШОЙ ТАНЕЦ =================
void danceRoutine() {
  if (!isDancing) return;

  // --- ЧАСТЬ 1: Разминка и радость ---
  setBodyTilt(90);
  setLEDColor(0, 255, 0); 
  moveArm(60);            
  if (waitAndCheck(500)) return;
  
  for (int i = 0; i < 4; i++) {
    rotateHead(50);
    moveRobot(200, -200);
    r2Chirp(2);
    if (waitAndCheck(1000)) return;

    rotateHead(-50);
    moveRobot(-200, 200);
    r2Chirp(2);
    if (waitAndCheck(1000)) return;
  }
  
  moveRobot(0, 0);
  moveArm(0); 
  if (waitAndCheck(1000)) return;

  // --- ЧАСТЬ 2: Поклоны и лунная походка ---
  setLEDColor(255, 0, 255); 
  for (int i = 0; i < 3; i++) {
    setBodyTilt(60); 
    moveRobot(-255, -255); 
    if (waitAndCheck(1500)) return;
    
    setBodyTilt(110); 
    moveRobot(255, 255); 
    r2Chirp(3);
    if (waitAndCheck(1000)) return;
  }
  
  moveRobot(0, 0);
  setBodyTilt(90);
  if (waitAndCheck(1000)) return;

  // --- ЧАСТЬ 3: Дискотека и болтовня головой ---
  for (int i = 0; i < 5; i++) {
    setLEDColor(random(0,255), random(0,255), random(0,255));
    rotateHead(random(-60, 60));
    moveArm(random(0, 60));
    
    moveRobot((i%2 == 0) ? 255 : -255, (i%2 == 0) ? -255 : 255);
    playSound(random(800, 2000), 100);
    if (waitAndCheck(500)) return;
    
    moveRobot(0, 0);
    if (waitAndCheck(500)) return;
  }

  // --- ЧАСТЬ 4: Завершающая стойка ---
  moveArm(60);
  rotateHead(0);
  setBodyTilt(90);
  setLEDColor(0, 255, 255); 
  r2Chirp(5);
  moveRobot(255, 255);
  if (waitAndCheck(800)) return;
  
  moveRobot(0, 0);
  if (waitAndCheck(3000)) return; // 3 секунды паузы перед новым кругом
  moveArm(0);
}

// ================= РЕАКЦИЯ НА СТОЛКНОВЕНИЕ (ГРУСТЬ) =================
void playEmotionShock() {
  moveRobot(0, 0); 
  
  // 1. Испуг
  setLEDColor(255, 0, 0); 
  rotateHead(45);
  tone(PIN_BUZZER, 400, 300); delay(300);
  rotateHead(-45);
  tone(PIN_BUZZER, 300, 300); delay(300);
  noTone(PIN_BUZZER);

  // 2. Уходим в грусть
  setBodyTilt(45); 
  rotateHead(0);   
  moveArm(0);      
  
  setLEDColor(0, 0, 100); 
  for(int freq = 1500; freq > 300; freq -= 30) {
    tone(PIN_BUZZER, freq, 50);
    delay(50);
  }
  noTone(PIN_BUZZER);

  // 3. Грустим 5 секунд
  // Во время грусти мы тоже опрашиваем веб-сервер!
  unsigned long sadnessStart = millis();
  while(millis() - sadnessStart < 5000) {
     server.handleClient();
     if (!isDancing) return; // Если нажали СТОП во время грусти - выходим
     delay(10);
  }

  // 4. Оживаем (если нас не выключили кнопкой)
  if (isDancing) {
    setBodyTilt(90); 
    setLEDColor(0, 255, 0); 
    tone(PIN_BUZZER, 2000, 200); 
    delay(250);
    noTone(PIN_BUZZER);
  }
}

// ================= БАЗОВЫЕ ФУНКЦИИ =================

void resetPosition() {
  moveRobot(0, 0);
  rotateHead(0);
  setBodyTilt(90);
  moveArm(0);
  setLEDColor(0, 0, 0);
  noTone(PIN_BUZZER);
}

void r2Chirp(int times) {
  for (int i = 0; i < times; i++) {
    setLEDColor(random(100,255), random(100,255), random(100,255));
    tone(PIN_BUZZER, random(1000, 3000), random(50, 150));
    delay(100);
    noTone(PIN_BUZZER);
  }
}

void moveRobot(int speedLeft, int speedRight) {
  if (speedLeft > 0) {
    analogWrite(IN1, speedLeft); analogWrite(IN2, 0);
  } else if (speedLeft < 0) {
    analogWrite(IN1, 0); analogWrite(IN2, abs(speedLeft));
  } else {
    analogWrite(IN1, 0); analogWrite(IN2, 0);
  }

  if (speedRight > 0) {
    analogWrite(IN3, speedRight); analogWrite(IN4, 0);
  } else if (speedRight < 0) {
    analogWrite(IN3, 0); analogWrite(IN4, abs(speedRight));
  } else {
    analogWrite(IN3, 0); analogWrite(IN4, 0);
  }
}

void rotateHead(int angle) {
  angle = constrain(angle, -60, 60);
  int servoAngle = map(angle, -60, 60, 30, 150);
  servoHead.write(servoAngle);
}

void setBodyTilt(int angle) {
  angle = constrain(angle, 45, 135);
  servoLegL.write(angle);
  servoLegR.write(180 - angle);
}

void moveArm(int position) {
  position = constrain(position, 0, 60);
  servoArm.write(position);
}

void setLEDColor(int r, int g, int b) {
  analogWrite(PIN_LED_R, r);
  analogWrite(PIN_LED_G, g);
  analogWrite(PIN_LED_B, b);
}

void playSound(unsigned int frequency, unsigned long duration) {
  tone(PIN_BUZZER, frequency, duration);
  delay(duration + 20);
  noTone(PIN_BUZZER);
}

bool checkCollision() {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);
  if (Wire.endTransmission(false) != 0) return false;
  
  Wire.requestFrom(MPU_ADDR, 6, true);
  if (Wire.available() >= 6) {
    int16_t rawX = (Wire.read() << 8) | Wire.read();
    int16_t rawY = (Wire.read() << 8) | Wire.read();
    int16_t rawZ = (Wire.read() << 8) | Wire.read();

    float totalAccel = sqrt((float)rawX * rawX + 
                            (float)rawY * rawY + 
                            (float)rawZ * rawZ);

    if (totalAccel > IMPACT_THRESHOLD) {
      return true;
    }
  }
  return false;
}
